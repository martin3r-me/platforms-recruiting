**Goal:** MA-Zeilen im ZAS-Export liefern Grundlohn = Tarif EG 1 (datiert) und einen Zuschlag, der an den RG-Gesamtlohn angleicht (+1 € bei Steuerklasse 6); der RG-Zuschlag kommt aus Bewerbung, sonst aus dem ZAS-Inbound (`zas_zuschlag`, main fff2ad69), sonst von HR; nur wirklich unbekannte RG-Löhne bleiben leer und landen in einer HR-Liste.

**Architecture:** Zwei reine Klassen (`EntgeltgruppeEins`, `LohnMa`) rechnen; ein Lese-Service (`LohnMaErmittlung`) findet die RG-Schwester und die Zuschlag-Quelle; der Export-Resolver ruft ihn im MA-Zweig. HR-Schreibtisch bekommt die Unterliste „Lohn", die MA-Akte einen Lese-Block, ein Kommando macht Vorflug und Marker.

```php
<?php

namespace Platform\Recruiting\Support;

/**
 * Entgeltgruppe 1 des Tarifwerks DGB/GVP (Stand Januar 2026), Anlage zum
 * Entgelttarifvertrag — der Grundlohn jeder MA-Anstellung (Markus 16.10.2026,
 * E15: "es gilt fuer alle Mitarbeiter EG 1").
 *
 * DATIERT, NICHT EINSTELLBAR: Der Tarif ist oeffentlich und hat feste Stufen.
 * Ein frei editierbarer Settings-Wert ohne Datum stuende im April 2027 still
 * falsch da. Neue Stufe = neue Zeile hier + Neu-Export der MA-Zeilen
 * (recruiting:lohn-ma-export --markieren), siehe Deploy-Checkliste.
 *
 * Reine Logik (kein Framework/DB).
 */
final class EntgeltgruppeEins
{
    /** @var list<array{ab:string, satz:float}> aufsteigend nach `ab` */
    public const STUFEN = [
        ['ab' => '2026-01-01', 'satz' => 14.96],
        ['ab' => '2026-09-01', 'satz' => 15.33],
        ['ab' => '2027-04-01', 'satz' => 15.87],
    ];

    /** Stundensatz am Stichtag (Y-m-d, Uhrzeit wird ignoriert); vor der ersten Stufe gilt die erste. */
    public static function satz(string $datum): float
    {
        return self::stufe($datum)['satz'];
    }

    /** Beginn der am Stichtag gueltigen Stufe (Y-m-d) — fuer Anzeige "EG 1 gilt seit". */
    public static function gueltigAb(string $datum): string
    {
        return self::stufe($datum)['ab'];
    }

    /** @return array{ab:string, satz:float} */
    private static function stufe(string $datum): array
    {
        $tag = substr($datum, 0, 10);
        $treffer = self::STUFEN[0];
        foreach (self::STUFEN as $stufe) {
            if ($stufe['ab'] <= $tag) {
                $treffer = $stufe;
            }
        }

        return $treffer;
    }
}
```

```php
<?php

namespace Platform\Recruiting\Support;

/**
 * Lohn einer MA-Anstellung fuer den ZAS-Export (Spec 2026-10-19 §3).
 *
 * Grundlohn = EG 1. Zuschlag gleicht an den RG-Gesamtlohn der Person an
 * (Markus 15.10.2026, E14: "mindestens den identischen Gesamtlohn"), dazu
 * +1,00 EUR bei Steuerklasse 6. Ohne RG-Anstellung gibt es nichts
 * anzugleichen ("nur MA ohne Zuschlag", 16.10.2026). Ist der RG-Zuschlag
 * unbekannt (Bestand ohne Bewerbung, nichts eingetragen), bleibt der
 * Zuschlag LEER — ein geratener Betrag landet im Lohn, ein leerer faellt auf.
 *
 * Reine Wertklasse, kein Framework/DB.
 */
final class LohnMa
{
    public const ANGLEICH  = 'angleich';
    public const ZAS       = 'zas';
    public const UNBEKANNT = 'unbekannt';
    public const MANUELL   = 'manuell';
    public const NUR_MA    = 'nur_ma';

    public const QUELLE_BEWERBUNG = 'bewerbung';
    public const QUELLE_ZAS       = 'zas';
    public const QUELLE_MANUELL   = 'manuell';

    public const KLASSE_SECHS_EURO = 1.00;

    private function __construct(
        public readonly string $fall,
        public readonly float $grundlohn,
        public readonly ?float $zuschlag,
        public readonly string $begruendung,
    ) {
    }

    /**
     * @param  float        $eg1          Tarifsatz EG 1 am Stichtag
     * @param  float        $mindestlohn  RG-Grundlohn (Team-Einstellung minimum_wage_hourly)
     * @param  bool         $hatRg        aktive RG-Anstellung derselben Person vorhanden
     * @param  float|null   $rgZuschlag   RG-Zuschlag; null = unbekannt
     * @param  string|null  $quelle       QUELLE_BEWERBUNG | QUELLE_ZAS | QUELLE_MANUELL | null
     * @param  string|null  $steuerklasse tax_class der MA-Anstellung ('1'..'6' oder null)
     */
    public static function rechne(float $eg1, float $mindestlohn, bool $hatRg, ?float $rgZuschlag, ?string $quelle, ?string $steuerklasse): self
    {
        $teile = ['EG 1 ' . self::euro($eg1)];
        $klasse = trim((string) $steuerklasse);
        $sechs = $klasse === '6' ? self::KLASSE_SECHS_EURO : 0.0;

        if (!$hatRg) {
            $teile[] = 'nur MA, kein Angleich';
            $zuschlag = round($sechs, 2);
            $fall = self::NUR_MA;
        } elseif ($rgZuschlag === null) {
            $teile[] = 'RG-Zuschlag unbekannt (keine Bewerbung, nichts eingetragen)';
            $teile[] = 'Zuschlag leer';

            return new self(self::UNBEKANNT, $eg1, null, implode(' · ', $teile));
        } else {
            $gesamt = $mindestlohn + $rgZuschlag;
            $angleich = max(0.0, $gesamt - $eg1);
            $herkunft = match ($quelle) {
                self::QUELLE_MANUELL => ', von HR eingetragen',
                self::QUELLE_ZAS     => ', laut ZAS',
                default              => '',
            };
            $teile[] = sprintf('Angleich an RG %s (%s + %s%s) → %s',
                self::euro($gesamt), self::zahl($mindestlohn), self::zahl($rgZuschlag), $herkunft, self::euro($angleich));
            $zuschlag = round($angleich + $sechs, 2);
            $fall = match ($quelle) {
                self::QUELLE_MANUELL => self::MANUELL,
                self::QUELLE_ZAS     => self::ZAS,
                default              => self::ANGLEICH,
            };
        }

        if ($sechs > 0) {
            $teile[] = 'Steuerklasse 6 +' . self::euro($sechs);
        } elseif ($klasse === '') {
            $teile[] = 'Steuerklasse unbekannt, kein +1 €';
        }
        $teile[] = 'Zuschlag ' . self::euro($zuschlag);

        return new self($fall, $eg1, $zuschlag, implode(' · ', $teile));
    }

    public function grundlohnText(): string
    {
        return self::zahl($this->grundlohn);
    }

    public function zuschlagText(): ?string
    {
        return $this->zuschlag === null ? null : self::zahl($this->zuschlag);
    }

    public static function zahl(float $v): string
    {
        return number_format($v, 2, ',', '.');
    }

    private static function euro(float $v): string
    {
        return self::zahl($v) . ' €';
    }
}
```

<?php

namespace Platform\Recruiting\Services;

use Illuminate\Support\Facades\DB;
use Platform\Recruiting\Models\RecApplicantSettings;
use Platform\Recruiting\Models\RecEmployee;
use Platform\Recruiting\Support\EntgeltgruppeEins;
use Platform\Recruiting\Support\LohnMa;
use Platform\Recruiting\Support\ZuschlagWert;

/**
 * Findet zu einer MA-Anstellung die RG-Schwester und die Quelle ihres
 * Zuschlags, und rechnet daraus den MA-Lohn (Spec Lohn MA 2026-10-19 §6).
 *
 * Lese-Service, Query Builder. Die RG-Schwester ist die AKTIVE RG-Anstellung
 * derselben Person (rec_person_id, sonst person_key ueber PersonScopeResolver);
 * eine ausgeschiedene RG-Anstellung zaehlt nicht — wer bei RG weg ist, ist
 * "nur MA". Bei mehreren die juengste.
 */
final class LohnMaErmittlung
{
    public function __construct(private readonly ?PersonScopeResolver $scope = null)
    {
    }

    public function ermitteln(RecEmployee $ma, ?string $datum = null): LohnMa
    {
        $tag = $datum ?? now()->toDateString();
        $rg = $this->rgSchwester($ma);
        $quelle = $rg === null ? ['wert' => null, 'quelle' => null] : $this->rgZuschlag($rg);

        return LohnMa::rechne(
            EntgeltgruppeEins::satz($tag),
            self::mindestlohn((int) $ma->team_id),
            $rg !== null,
            $quelle['wert'],
            $quelle['quelle'],
            $ma->tax_class !== null ? (string) $ma->tax_class : null,
        );
    }

    /** @return object|null Zeile mit id, rec_applicant_id, zuschlag_manuell, zas_grundlohn, zas_zuschlag */
    public function rgSchwester(RecEmployee $ma): ?object
    {
        $q = $this->rgAbfrage()
            ->where('e.id', '!=', (int) $ma->id);

        if ($ma->rec_person_id !== null) {
            $q->where('e.rec_person_id', (int) $ma->rec_person_id);
        } else {
            $ids = array_values(array_diff(($this->scope ?? new PersonScopeResolver())->forEmployee($ma)['ids'], [(int) $ma->id]));
            if ($ids === []) {
                return null;
            }
            $q->whereIn('e.id', $ids);
        }

        return $q->orderByDesc('e.id')->first();
    }

    /** Dieselbe Zeile fuer eine RG-Anstellung selbst (RG-Export ohne Bewerbung). */
    public function rgZeile(int $employeeId): ?object
    {
        return $this->rgAbfrage()->where('e.id', $employeeId)->first();
    }

    /** Aktive RG-Anstellungen mit ihren ZAS-Lohnwerten (Left Join, hr_data kann fehlen). */
    private function rgAbfrage(): \Illuminate\Database\Query\Builder
    {
        return DB::table('rec_employees as e')
            ->leftJoin('rec_employee_hr_data as h', 'h.rec_employee_id', '=', 'e.id')
            ->where('e.is_active', 1)
            ->where(fn ($w) => $w->whereNull('e.company')->orWhere('e.company', '')->orWhereRaw('UPPER(TRIM(e.company)) = ?', ['RG']))
            ->select(['e.id', 'e.rec_applicant_id', 'e.zuschlag_manuell', 'h.zas_grundlohn', 'h.zas_zuschlag']);
    }

    /**
     * Zuschlag der RG-Anstellung in dieser Reihenfolge: Bewerbung (Datenfeld,
     * sonst AV-Code, sonst 0,00 — Bewerbung vorhanden heisst bekannt), sonst
     * ZAS (zas_zuschlag, aber nur bei zas_grundlohn > 0: Grundlohn 0 heisst
     * bei ZAS "Zeile nicht gepflegt", der Zuschlag daneben ist keine Angabe —
     * Befund 19.10.2026, 125 solcher Zeilen), sonst zuschlag_manuell, sonst
     * unbekannt. Dieselbe Kette fuer MA-Angleich und RG-Export.
     *
     * @return array{wert:?float, quelle:?string}
     */
    public function rgZuschlag(object $rg): array
    {
        if ($rg->rec_applicant_id !== null) {
            $z = DB::table('rec_applicants')->where('id', (int) $rg->rec_applicant_id)->value('zuschlag');
            if ($z === null) {
                $code = DB::table('rec_applicants')
                    ->join('rec_contract_templates', 'rec_applicants.contract_template_id', '=', 'rec_contract_templates.id')
                    ->where('rec_applicants.id', (int) $rg->rec_applicant_id)
                    ->value('rec_contract_templates.code');
                $z = ZuschlagWert::ausAvCode($code !== null ? (string) $code : null) ?? 0.0;
            }

            return ['wert' => (float) $z, 'quelle' => LohnMa::QUELLE_BEWERBUNG];
        }
        if ($rg->zas_zuschlag !== null && (float) $rg->zas_grundlohn > 0) {
            return ['wert' => (float) $rg->zas_zuschlag, 'quelle' => LohnMa::QUELLE_ZAS];
        }
        if ($rg->zuschlag_manuell !== null) {
            return ['wert' => (float) $rg->zuschlag_manuell, 'quelle' => LohnMa::QUELLE_MANUELL];
        }

        return ['wert' => null, 'quelle' => null];
    }

    /**
     * Aktive MA-Anstellungen derselben Person, ohne die eigene — fuer den
     * Re-Export-Marker, wenn sich der RG-Zuschlag aendert.
     *
     * @return list<int>
     */
    public static function maSchwesternIds(int $employeeId, ?int $personId, ?string $personKey): array
    {
        $q = DB::table('rec_employees')->where('id', '!=', $employeeId)->where('is_active', 1)
            ->whereRaw('UPPER(TRIM(company)) = ?', ['MA']);
        if ($personId !== null) {
            $q->where('rec_person_id', $personId);
        } elseif (trim((string) $personKey) !== '') {
            $q->where('person_key', trim((string) $personKey));
        } else {
            return [];
        }

        return $q->orderBy('id')->pluck('id')->map(fn ($v) => (int) $v)->all();
    }

    public static function mindestlohn(int $teamId): float
    {
        $wage = RecApplicantSettings::getOrCreateForTeam($teamId)->getSetting('minimum_wage_hourly');

        return $wage !== null ? (float) $wage : 13.90;
    }

    public static function istMa(?string $company): bool
    {
        return strtoupper(trim((string) $company)) === 'MA';
    }
}

```php
class LohnMaExport extends Command
{
    protected $signature = 'recruiting:lohn-ma-export
        {--markieren : zas_changed_at auf alle Zeilen des Laufs setzen (Neu-Export)}
        {--team= : Nur dieses Team}
        {--datum= : Stichtag fuer EG 1 (Y-m-d, Default heute)}

    protected $description = 'MA-Loehne (EG 1 + Angleich) je Anstellung rechnen, zaehlen, als CSV ablegen; optional fuer ZAS markieren';

    public function handle(): int
    {
        $markieren = (bool) $this->option('markieren');
        $team = $this->option('team') !== null ? (int) $this->option('team') : null;
        $datum = $this->option('datum') !== null ? (string) $this->option('datum') : null;

            : storage_path('app/recruiting/lohn-ma-' . ($datum ?? now()->toDateString()) . '-' . now()->format('His') . '.csv');

            match ($typ) {
                'warn' => $this->warn($text),
                default => $this->line($text),
            };
        });

        $this->newLine();
        foreach ($r['zahlen'] as $fall => $n) {
            $this->line(sprintf('%-10s %5d', $fall, $n));
        }

        $q = RecEmployee::query()->where('is_active', 1)->whereRaw('UPPER(TRIM(company)) = ?', ['MA'])->orderBy('id');
```
